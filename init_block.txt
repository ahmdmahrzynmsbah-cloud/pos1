    fun checkpointDatabase() {
        try {
            val cursor = database.query(androidx.sqlite.db.SimpleSQLiteQuery("pragma wal_checkpoint(full)"))
            cursor.close()
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    fun closeDatabase() {
        try {
            if (database.isOpen) {
                database.close()
            }
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }
